	/**
	* Given a path, adds/overwrites that editor to local storage if it is a valid
	* editor and version is found.
	* It set the default editor if this is the first editor added.
	* @param {string} location
	* @param {boolean} skipSignatureCheck Whether to skip Unity signature checking
	* @returns {Promise<LocatedInstall>}
	* @public
	*/
	addEditor(location, skipSignatureCheck) {
		return this.validateEditorFile(location, skipSignatureCheck).then(() => this.findVersion(location)).then(async (versionInfo) => {
			const editorArchitecture = await this.findArchitecture(location);
			const productName = await editorUtils.findUnityProductName(path.dirname(location)) || getProductNameFallback(versionInfo.version);
			const newEditor = {
				version: versionInfo.version,
				location: [location],
				manual: true,
				architecture: editorArchitecture,
				productName
			};
			if (find(this.editorList.availableEditors, {
				version: newEditor.version,
				architecture: newEditor.architecture
			})) return Promise.reject({
				errorCode: INSTALL_LOCATE_ERRORS.ALREADY_IN_LIST,
				params: { version: newEditor.version }
			});
			return this.editorList.addLocatedEditor(newEditor).then(() => {
				cloudAnalytics.addEvent({
					type: cloudAnalytics.eventTypes.EDITOR_LOCATE,
					msg: {
						editor_version: newEditor.version,
						editor_architecture: newEditor.architecture,
						used_shift_key: skipSignatureCheck,
						status: "Success"
					}
				});
			}).then(() => newEditor).catch((err) => {
				cloudAnalytics.addEvent({
					type: cloudAnalytics.eventTypes.EDITOR_LOCATE,
					msg: {
						editor_version: newEditor.version,
						editor_architecture: newEditor.architecture,
						used_shift_key: skipSignatureCheck,
						status: "Error"
					}
				});
				return Promise.reject(err);
			});
		});
	}

	* @param {Install} editor - The editor to uninstall.
	* @returns {Promise<void | { cancelled: boolean }>}
	* @public

	async uninstallEditor(editor) {
		try {
			if ((await unityInstaller.prepareInstallService())?.cancelled) return { cancelled: true };
			if ((await Promise.all(editor.location.map((editorPath) => unityInstaller.uninstall(editor, new EditorApp({
				editorPath,
				version: editor.version,
				architecture: editor.architecture
			}).path)))).some((response) => response?.cancelled)) return { cancelled: true };
			await this.editorList.removeInstalledEditor(editor);
		} catch (error) {
			throw new Error(`The Editor ${getUniqueIdentifier(editor)} could not be uninstalled.`, { cause: error });
		}
	}

	/**
	* Automatically locate the editor that is connected to the hub
	* @param {string} location
	* @returns {Promise<void | LocatedInstall>}
	* @public

	async checkAutoLocate(location) {
		const { version } = await this.findVersion(location);
		const architecture = await this.findArchitecture(location);
		const editorWithTheSameVersion = this.availableEditors[getUniqueIdentifier({
			version,
			architecture
		})];
		if (!editorWithTheSameVersion || editorWithTheSameVersion.location.indexOf(location) < 0) return this.addEditor(location, true).catch((err) => Promise.reject(err));
		return Promise.resolve();
	}
	/**
	* Get/find an editor install given a set of criteria to match exactly in the install props.
	*
	* @param {{ version: string } & Partial<Install>} criteria The criteria for finding the editor. Any props on the install can be
	* selected in the criteria, but only using strict equality (i.e. object values won't work).
	* @param {boolean} [installationFlow=false] Whether the call is part of the installation flow.
	* @returns {Install} The found editor install.
	* @public
	*/
	getEditor(criteria, installationFlow = false) {
		const editors = Object.values(this.availableEditors);
		let editor = find(editors, criteria);
		let multipleEditorsFound = false;
		if (!installationFlow) {
			const editorsWithMatchingVersion = this.findEditorsByVersion(criteria.version);
			if (editorsWithMatchingVersion.length === 1) editor = editorsWithMatchingVersion[0];
			else if (editorsWithMatchingVersion.length === 2) multipleEditorsFound = true;
		}
		if (editor) return editor;
		if (multipleEditorsFound) throw new EditorError(`Multiple editors found matching the version ${criteria.version} and no architecture was specified`, this.errorCodes.MULTIPLE_EDITOR_FOUND);
		else if (editors.length === 0) throw new EditorError(`No editor installed. Could not be retrieved for criteria: ${JSON.stringify(criteria)}`, this.errorCodes.NO_EDITORS_FOUND);
		throw new EditorError(`Editor could not be retrieved for criteria: ${JSON.stringify(criteria)}`, this.errorCodes.EDITOR_NOT_FOUND);
	}
	/**
	* Get/find list of editors install given a version to match exactly in the install props.

	* @returns The list of found editor install.
	* @public

	findEditorsByVersion(version) {
		const foundEditors = [];
		Object.keys(this.availableEditors).forEach((key) => {
			if (key.includes(version)) foundEditors.push(this.availableEditors[key]);
		});

	}

	/**
	* Download and installation could be done on 2 different disks
	* If space is missing for either, we should warn the user and reject the promise.
	* @param editor Information about the current editor to install or to add modules to. Knows where the installs will be
	* @param modules Information about the modules to install. Could be empty
	* @param downloadDest Where to download the required files
	* @param onlyModules Whether we should add the dl/install size of the editor or not to the disk size required
	*/
	hasEnoughSpaceOnDisk(editor, modules, downloadDest, onlyModules) {
		return this.getRequiredSizesPerDisk(editor, modules, downloadDest, onlyModules).then((diskList) => {
			for (const disk of Object.keys(diskList)) if (diskList[disk].available < diskList[disk].totalSizeRequired) return Promise.reject("Not enough space on disk to download and install");
			return Promise.resolve();
		});
	}
	getRequiredSizesPerDisk(editor, modules, downloadDest, onlyModules) {
		const diskInfo = {};
		let downloadDiskRootPath = "";
		let installDiskRootPath = "";
		try {
			downloadDiskRootPath = hub_fs_default.getDiskRootPath(downloadDest);
			const editorPath = editor.location ? editor.location[0] : this.currentInstallPath;
			installDiskRootPath = hub_fs_default.getDiskRootPath(editorPath);
		} catch (e) {
			logger$2.error("Could not validate size on disk for download because of ", e);
			return Promise.resolve();
		}
		return hub_fs_default.getDiskUsage(downloadDiskRootPath).then((info) => {
			diskInfo[downloadDiskRootPath] = {
				totalSizeRequired: 0,
				available: info.available
			};
		}).then(() => hub_fs_default.getDiskUsage(installDiskRootPath)).then((info) => {
			if (!diskInfo[installDiskRootPath]) diskInfo[installDiskRootPath] = {
				totalSizeRequired: 0,
				available: info.available
			};
		}).then(() => {
			forEach(modules, (module) => {
				diskInfo[downloadDiskRootPath].totalSizeRequired += parseInt(module.downloadSize, 10);
				diskInfo[installDiskRootPath].totalSizeRequired += parseInt(module.installedSize, 10);
			});
			if (!onlyModules) {
				diskInfo[downloadDiskRootPath].totalSizeRequired += parseInt(editor.downloadSize, 10);
				diskInfo[installDiskRootPath].totalSizeRequired += parseInt(editor.installedSize, 10);
			}
			return Promise.resolve(diskInfo);
		}).catch((e) => {
			logger$2.error("Could not calculate required size on disk for download because of", e);
			return Promise.resolve({});
		});
	}

	/**
	* Get the default modules for an editor.
	*
	* @param editorUniqueId
	* The unique identification of the editor that default modules should be returned for.
	* @returns The modules list from the
	* `modules.json`, CDN, or releases file for the given editor.
	* @public
	*/
	async getEditorDefaultModules(editorUniqueId) {
		const filteredEditorUniqueId = pick(editorUniqueId, ["version", "architecture"]);
		try {
			const installedEditor = this.getEditor(filteredEditorUniqueId, true);
			if (installedEditor.revisionHash) {
				logger$2.info(`Editor has revisionHash ${installedEditor.revisionHash}, fetching modules from CDN`);
				const cdnRelease = await fetchEditorReleaseFromCdn(installedEditor.version, installedEditor.revisionHash, installedEditor.architecture);
				if (cdnRelease?.modules) {
					logger$2.info(`Successfully fetched ${cdnRelease.modules.length} modules from CDN for ${installedEditor.version}`);
					const installedModules = installedEditor.modules ?? [];
					return cdnRelease.modules.map((cdnModule) => {
						const installedModule = installedModules.find((m) => m.id === cdnModule.id);
						return {
							...cdnModule,
							selected: installedModule?.selected ?? false
						};
					});
				}
				logger$2.warn(`Failed to fetch CDN modules for ${installedEditor.version}, falling back to installed modules`);
			}
			return installedEditor.modules ?? [];
		} catch (_e) {
			return this.getDefaultReleasesModules(filteredEditorUniqueId);
		}
	}
