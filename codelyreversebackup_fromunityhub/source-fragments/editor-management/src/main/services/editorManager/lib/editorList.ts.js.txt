var logger$3 = loggerFactory("EditorList");
var EditorList = class extends EventEmitter {
	get latest() {
		const editorKeys = Object.keys(this.availableEditors);
		let latest = null;
		if (editorKeys.length > 0) {
			editorKeys.sort((version1, version2) => {
				if (version1.split("-")[0] === version2.split("-")[0]) return version1.split("-")[1] === CHIP_ARCHITECTURES.ARM64 ? 1 : -1;
				version1 = version1.split("-")[0];
				version2 = version2.split("-")[0];
				if (e$4.isValid(version1) === false) return -1;
				else if (e$4.isValid(version2) === false) return 1;
				return new e$4(version1).compare(version2);
			});
			latest = this.availableEditors[editorKeys[editorKeys.length - 1]];
		}
		return latest;
	}
	get secondaryInstallPath() {
		return this.#installedEditorList.secondaryInstallPath;
	}
	#installedEditorList;
	#locatedEditorStorage;
	availableEditors;
	constructor(installPath) {
		super();
		this.#installedEditorList = new InstalledEditorList(installPath);
		this.#locatedEditorStorage = new LocatedEditorStorage();
		this.availableEditors = {};
	}
	init() {
		logger$3.info("init");
		return Promise.resolve().then(() => this.#installedEditorList.findInstalledEditors()).then(() => this.#locatedEditorStorage.fetchEditors()).then(() => this.#updateAvailableEditors());
	}
	async initEditorsBuildPlatform() {
		await Promise.all(Object.values(this.availableEditors).map((editor) => this.addBuildPlatformsToEditor(editor)));
	}
	async addBuildPlatformsToEditor(editor) {
		editor.buildPlatforms = await editorUtils.findBuildPlatforms(editor.location[0]);
	}
	async initUlfRequirementCheck() {
		const localSettings = mainContainer.get(ILocalSettingsId);
		const isFloatingEnabled = Boolean(localSettings.get(localSettings.keys.globalMachineSettings.LICENSING_SERVICE_BASE_URL));
		await Promise.all(Object.values(this.availableEditors).map((editor) => this.addUlfRequirementFlag(editor, isFloatingEnabled)));
	}
	async addUlfRequirementFlag(editor, isFloatingEnabled) {
		editor.requiresUlf = !isFloatingEnabled && this.doesVersionRequireUlf(editor.version);
	}
	doesVersionRequireUlf(version) {
		return [
			compareEditorVersions("2019.4.27f1", version) === -1,
			compareEditorVersions("2020.1.0a1", version) >= 0 && compareEditorVersions("2020.1.17f1", version) <= 0,
			compareEditorVersions("2020.2.0a1", version) >= 0 && compareEditorVersions("2020.2.2f1", version) === -1
		].some((condition) => condition);
	}
	async #updateAvailableEditors() {
		this.availableEditors = Object.assign({}, this.#locatedEditorStorage.editors, this.#installedEditorList.editors);
		await this.initEditorsBuildPlatform();
		await this.initUlfRequirementCheck();
		this.emit(INSTALL_LIST_EVENTS.AVAILABLE_EDITORS_CHANGED);
		windowManager_default.broadcastContent(RENDERER_EVENTS.AVAILABLE_EDITORS_UPDATE, this.availableEditors);
	}
	hasEditorWhichRequiresUlf() {
		return Object.values(this.availableEditors).some((editor) => editor.requiresUlf === true);
	}
	addLocatedEditor(editor) {
		return Promise.resolve().then(() => {
			if (this.availableEditors[getUniqueIdentifier(editor)]) throw new Error(`Another Unity ${editor.version} exists on your machine with the same architecture.
         You need to uninstall/remove it before adding a new one.`);
			return this.#locatedEditorStorage.addEditor(editor);
		}).then(() => this.#updateAvailableEditors()).then(() => this.availableEditors[getUniqueIdentifier(editor)]);
	}
	removeLocatedEditor(editor) {
		return this.#locatedEditorStorage.removeEditor(editor).then(() => this.#updateAvailableEditors());
	}
	async addInstalledEditor(editor) {
		await this.#installedEditorList.addEditor(editor);
		return this.#updateAvailableEditors();
	}
	async updateInstalledEditor(editor) {
		await this.#installedEditorList.updateEditor(editor);
		return this.#updateAvailableEditors();
	}
	removeInstalledEditor(editor) {
		this.#installedEditorList.removeEditor(editor);
		return this.#updateAvailableEditors();
	}
	setSecondaryInstallLocation(installPath) {
		return this.#installedEditorList.setSecondaryInstallLocation(installPath).then(() => this.#updateAvailableEditors());
	}
};
