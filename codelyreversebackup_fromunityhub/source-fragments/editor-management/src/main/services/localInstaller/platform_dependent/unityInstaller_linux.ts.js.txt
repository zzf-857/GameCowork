async function findPayloadPath(directory) {
	const files = await fs.promises.readdir(directory);
	const pkgTmpDir = files.find((file) => file.endsWith(".pkg.tmp"));
	if (pkgTmpDir) {
		logger.debug(`Found .pkg.tmp directory: ${pkgTmpDir}`);
		const payloadDir = path.join(directory, pkgTmpDir);
		const payloadTildePath = path.join(payloadDir, "Payload~");
		if (await fs.promises.access(payloadTildePath).then(() => true).catch(() => false)) {
			logger.debug("Found Payload~ in .pkg.tmp directory (gzip already extracted)");
			return payloadTildePath;
		}
		return path.join(payloadDir, "Payload");
	}
	const payloadFile = files.find((file) => file === "Payload~");
	if (payloadFile) {
		logger.debug(`Found Payload~ file: ${payloadFile}`);
		return path.join(directory, payloadFile);
	}
	logger.warn("Payload file not found in pkg archive");
	throw new InstallerError(INSTALLER_ERROR.COULD_NOT_EXTRACT, { cause: /* @__PURE__ */ new Error("Payload file not found in pkg archive") });
}
