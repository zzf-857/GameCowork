function decodeError(installError) {
	if (installError === void 0 || installError === null || typeof installError !== "object") return INSTALLER_ERROR.DEFAULT;
	if ("code" in installError && typeof installError.code === "string") {
		const errorCode = installError.code;
		if (INSTALLER_ERROR[errorCode]) return INSTALLER_ERROR[errorCode];
	}
	if ("message" in installError && typeof installError.message === "string") {
		const errorMessage = installError.message;
		const error = Object.values(INSTALLER_ERROR).find((error) => error.message === errorMessage);
		if (error) return error;
	}
	if ("error" in installError && typeof installError.error === "object") return decodeError(installError.error);
	return INSTALLER_ERROR.DEFAULT;
}
var InstallerError = class extends ErrorBase {
	code;
	error;
	constructor(error, options = {}) {
		super(error.message, {
			name: error.code,
			cause: options.cause
		});
		this.code = error.code;
		this.error = error;
	}
};
