/**

*/

*/

	X86_64: "X86_64",
	ARM64: "ARM64"
};

/**
* Maps CDN module to Hub's InstallModule format.
*/
function mapCdnModuleToInstallModule(module, revisionHash, parent = "") {
	const mappedModules = [];
	const fullDownloadUrl = module.url ? module.url.startsWith("http://") || module.url.startsWith("https://") ? module.url : revisionHash ? `https://download.unity3d.com/download_unity/${revisionHash}/${module.url}` : module.url : "";
	const installModule = {
		id: module.id,
		name: module.name,
		description: module.description,
		downloadUrl: fullDownloadUrl,
		category: module.category,
		installedSize: module.installedSize?.value ?? 0,
		downloadSize: module.downloadSize?.value ?? 0,
		visible: !module.hidden,
		selected: module.preSelected ?? false,
		preselected: module.preSelected ?? false,
		parent,
		sync: module.hidden && module.required ? parent : module.sync ?? "",
		destination: module.destination ?? "",
		eulaUrl1: module.eula?.[0]?.url ?? "",
		eulaLabel1: module.eula?.[0]?.label ?? "",
		eulaMessage: module.eula?.[0]?.message ?? "",
		renameFrom: module.extractedPathRename?.from ?? "",
		renameTo: module.extractedPathRename?.to ?? "",
		integrity: module.checksum ?? module.integrity ?? ""
	};
	mappedModules.push(installModule);
	if (module.subModules) for (const subModule of module.subModules) mappedModules.push(...mapCdnModuleToInstallModule(subModule, revisionHash, module.id));
	return mappedModules;
}
