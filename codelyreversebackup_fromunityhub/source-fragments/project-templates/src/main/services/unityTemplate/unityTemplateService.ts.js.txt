	async _checkTemplateFileIntegrity({ templatesPath, templateName, templateVersion, integrity }) {

		const hash = createHash("sha1");
		const templateReadStream = import_lib$4.default.createReadStream(templatePath);
		return new Promise((resolve, reject) => {

			remoteVersion: versionName,
			remoteTar: tarball.url,
			integrity: tarball.integrity,
			size: size.value,
			remoteSize: size.value,
			packages: dependencies,
			dynamicIcon: icon.url,
			previewImage: previewImage.url,
			renderPipeline,
			buildPlatforms,
			minimumAge: ageGateField ? ageGateField.value : null,
			termsOfServiceUrl: termsOfService ? termsOfService.url : null,
			previousVersions: versions.slice(1)
		};
	}
	_setTemplateStatus(status) {
		return (template) => ({
			...template,
			status
		});
	}
	/**
	* Run a crude validation of a built-in template, ensuring the minimum required fields are present
	* and the correct types.
	*
	* @param param0 An object possibly containing the minimum required template fields.
	* @returns `true` if the supplied template has the minimum required fields.
	*/
	_isValidTemplate({ name, displayName, version }) {
		return typeof name === "string" && name.length > 0 && typeof displayName === "string" && displayName.length > 0 && typeof version === "string" && version.length > 0;
	}
	/**
	* Convert an array of templates to an object mapping of template names to templates.
	*
	* @param templates The array of
	* templates to create an object mapping for.
	* @returns The object mapping of template names to templates.
	*/
	_templatesToTemplateMap(templates) {
		return templates.reduce((map, template) => {
			map[template.name] = template;
			return map;
		}, {});
	}
	/**
	* Remove stale cached templates for editor versions that are not currently known by Hub.
	*/
	_filterStaleTemplatesCache() {
		const installedVersions = Object.values(editorManager_default.availableEditors).map(({ version }) => version);
		this.#unityTemplateCache.filter((editorVersion) => installedVersions.includes(editorVersion));
	}
};
var unityTemplateService_default = new Templates();
