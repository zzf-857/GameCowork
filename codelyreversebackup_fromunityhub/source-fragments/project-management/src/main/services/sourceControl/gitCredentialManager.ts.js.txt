var USER_INFO_KEY = "userInfoKey";
var UserInfoService = class {
	logger = loggerFactory("UserInfoService");
	async getUserInfoFromStorage() {
		if (!safeStorage.isEncryptionAvailable()) {
			this.logger.warn("Safe storage is not available, user info will not be retrieved");
			return;
		}
		try {
			const jsonStorage = mainContainer.get(IJsonStorageId);
			if (await jsonStorage.has("userInfoKey")) {
				const res = await jsonStorage.get(USER_INFO_KEY);
				const bufferRes = Buffer.from(res.data, "latin1");
				return JSON.parse(safeStorage.decryptString(bufferRes));
			}
			return;
		} catch (e) {
			this.logger.debug(`Could not get user info from ${USER_INFO_KEY}.json file`, e);
			return;
		}
	}
	async setUserInfoToStorage(user) {
		if (!safeStorage.isEncryptionAvailable()) {
			this.logger.warn("Safe storage is not available, user info will not be stored");
			return;
		}
		try {
			const userInfoEncryptedObject = { data: safeStorage.encryptString(JSON.stringify(user)).toString("latin1") };
			await mainContainer.get(IJsonStorageId).set(USER_INFO_KEY, userInfoEncryptedObject);
		} catch (error) {
			this.logger.error("Unable to store user info", error);
			throw error;
		}
	}
	async clearUserInfoFromStorage() {
		try {
			await mainContainer.get(IJsonStorageId).remove(USER_INFO_KEY);
		} catch (error) {
			this.logger.error("Unable to clear user info.", error);
			throw error;
		}
	}
};
