					logger$28.info(`Also added socket to deeplinkReadyAvailableIPCSockets for project "${singleOpenProjectPath}" (editor supports project.openInEditorUrl)`);
				}
			}
			this.openProjectsCache.clear();
		} else if (openProjects.length > 1) {
			logger$28.warn(`Cannot determine which project to associate with incoming socket: ${openProjects.length} projects are currently open (${openProjects.join(", ")})`);
			this.openProjectsCache.clear();
		} else {
			logger$28.info(`No open projects found in cache, discarding socket mapping`);
			this.openProjectsCache.clear();
		}
	}
	/**
	* Rebuilds the list of local project linked to an Editor that supports deeplinking.
	*
	* @param forceRefresh - When true, forces a refresh of project data from disk regardless of window focus state. Defaults to false.
	* @param postalTopic - The postal topic to validate support for. Defaults to PROJECT_OPEN_IN_EDITOR_URL for regular deeplinks.
	*
	*/
	async refreshDeeplinkEnabledLocalProjects(forceRefresh = false, postalTopic = POSTAL_TOPICS.PROJECT_OPEN_IN_EDITOR_URL) {
		if (forceRefresh) await this.getRecentProjects(true);
		this.deeplinkEnabledLocalProjectIdSet.clear();
		await Promise.all(Array.from(this.projects.values()).map(async (projectEntity) => {
			const versionArchTopicKey = `${projectEntity.version}-${projectEntity.architecture}-${postalTopic}`;
			let projectEditorVersionValidated = this.deeplinkSupportEditorVersionCache.has(versionArchTopicKey);
			if (!projectEditorVersionValidated) {
				projectEditorVersionValidated = await this.validateEditorHasIPCFeatureSupport(projectEntity.version, projectEntity.architecture, postalTopic);
				this.deeplinkSupportEditorVersionCache.set(versionArchTopicKey, projectEditorVersionValidated);
			}
		}));
		for (const [_path, projectEntity] of this.projects) {
			const versionArchTopicKey = `${projectEntity.version}-${projectEntity.architecture}-${postalTopic}`;
			if (this.deeplinkSupportEditorVersionCache.get(versionArchTopicKey)) this.deeplinkEnabledLocalProjectIdSet.add(projectEntity.localProjectId);
		}
		windowManager_default.broadcastContent(RENDERER_EVENTS.DEEPLINK_ENABLED_PROJECT_LIST_UPDATE, [...this.deeplinkEnabledLocalProjectIdSet]);
	}

	/**
	* Returns if an editor supports the specified IPC feature
	*
	* @access public
	* @param {string} editorVersion the editor version
	* @param {ChipArchitecture} editorArchitecture the editor chip architecture
	* @param {PostalTopic} postalTopic the postal topic to validate (either POSTAL_TOPICS.PROJECT_OPEN_IN_EDITOR_URL or POSTAL_TOPICS.SHOW_PROJECT_MANAGER)
	* @returns {Promise<boolean>}
	*/
	async validateEditorHasIPCFeatureSupport(editorVersion, editorArchitecture, postalTopic) {
		const editorApp = await editorManager_default.getEditorApp({
			version: editorVersion,
			architecture: editorArchitecture
		});
		if (editorApp.version !== editorVersion) return false;
		const ipcFeaturesPath = editorApp.exec;
		const ipcFeaturesJsonRelativePath = getUserOS() === PLATFORMS.MAC ? "../Resources/PackageManager/IPC/ipcFeatures.json" : "Data/Resources/PackageManager/IPC/ipcFeatures.json";
		const combinedPath = path.resolve(path.dirname(ipcFeaturesPath), ipcFeaturesJsonRelativePath);
		if (!await hub_fs_default.isFilePresentAsync(combinedPath)) {
			logger$28.warn(`ipcFeatures.json does not exist: ${combinedPath}`);
			return false;
		}
		let hasIPCFeature = false;
		let canReadFile = false;
		let content;
		try {
			content = await fs.promises.readFile(combinedPath, "utf8");
			canReadFile = true;
		} catch (err) {
			logger$28.warn(`Failed to read file ${err}`);
			return false;
		}
		if (canReadFile) try {
			const json = JSON.parse(content);
			if (Array.isArray(json.ipc) && json.ipc.includes(postalTopic)) {
				logger$28.warn(`${postalTopic} ipc message is supported.`);
				hasIPCFeature = true;
			} else logger$28.warn(`${postalTopic} ipc message is not supported.`);
		} catch (err) {
			logger$28.warn(`Error parsing json ${err}`);
		}
		return hasIPCFeature;
	}

	/**
	* Adds or updates a project in the recent projects list.
	*
	* This method is called when a project is opened, created, or otherwise accessed.
	* It updates the project's last modified time, refreshes project metadata,
	* applies any provided cloud project information, and ensures the project
	* appears at the top of the recent projects list.
	*
	* @param projectPath - The absolute path to the Unity project directory
	* @param editorVersion - The Unity editor version associated with the project
	* @param editorArchitecture - The required architecture for the project
	* @param cloudProjectId - Optional Unity Cloud project ID
	* @param organizationId - Optional Unity organization ID
	* @param genesisOrgId - Optional Genesis organization ID
	* @returns Promise that resolves to the updated project data
	*/
	async addToRecentProjects(projectPath, editorVersion, editorArchitecture, cloudProjectId, organizationId, genesisOrgId) {
		let projectData = this.projects.get(projectPath);
		const now = Date.now();
		try {
			await fs.promises.utimes(projectPath, now / 1e3, now / 1e3);
		} catch (e) {
			logger$28.warn(`Failed to update folder mtime for project at ${projectPath}:`, e);
		}
		if (projectData) projectData.lastModified = now;
		else {
			projectData = await this.getProjectData(projectPath);
			try {
				const vcsData = await this.vcsManagementService.getProjectVCSData(projectPath);
				if (vcsData) {
					projectData.organizationName = vcsData.organizationName;
					projectData.repositoryName = vcsData.repositoryName;
					projectData.projectPathInsideRepository = vcsData.projectPathInsideRepository;
					projectData.vcsConfigurationPath = vcsData.vcsConfigurationPath;
					projectData.vcsProvider = vcsData.provider;
				}
			} catch (error) {
				logger$28.debug("VCS detection failed for newly added project", {
					projectPath,
					error
				});
			}
			if (!projectData.lastModified) projectData.lastModified = now;
		}
		if (cloudProjectId) projectData.cloudProjectId = cloudProjectId;
		if (editorVersion) {
			projectData.version = editorVersion;
			if (editorArchitecture) projectData.architecture = editorArchitecture === CHIP_ARCHITECTURES.Unknown ? this.deriveArchitecture(editorVersion) : editorArchitecture;
		}
		if (organizationId) projectData.organizationId = organizationId;
		if (genesisOrgId) projectData.genesisOrgId = genesisOrgId;
		this.projects.set(projectPath, projectData);
		await this.saveProjects();
		windowManager_default.broadcastContent(RENDERER_EVENTS.PROJECT_LIST_UPDATE, this.getProjectsFromMemory());
		return projectData;
	}

	/**
	* Updates project metadata after file system changes are detected.
	*
	* This method refreshes a project's metadata by re-reading its configuration files
	* and updating the in-memory project cache. It also saves the changes to persistent
	* storage and broadcasts updates to the UI.
	*
	* @param projectPath - The absolute path to the Unity project directory
	* @returns Promise that resolves when the project data is updated
	*/
	async updateProjectSettings(projectPath, options) {
		try {
			const existingProjectData = this.projects.get(projectPath);
			const newProjectData = await this.getProjectData(projectPath);
			if (options?.skipVCSDetection) {
				if (existingProjectData) {
					newProjectData.organizationName = existingProjectData.organizationName;
					newProjectData.repositoryName = existingProjectData.repositoryName;
					newProjectData.projectPathInsideRepository = existingProjectData.projectPathInsideRepository;
					newProjectData.vcsConfigurationPath = existingProjectData.vcsConfigurationPath;
					newProjectData.vcsProvider = existingProjectData.vcsProvider;
				}
			} else if (options?.vcsData !== void 0) {
				const vcsData = options.vcsData;
				if (vcsData) {
					newProjectData.organizationName = vcsData.organizationName;
					newProjectData.repositoryName = vcsData.repositoryName;
					newProjectData.projectPathInsideRepository = vcsData.projectPathInsideRepository;
					newProjectData.vcsConfigurationPath = vcsData.vcsConfigurationPath;
					newProjectData.vcsProvider = vcsData.provider;
				} else {
					newProjectData.organizationName = void 0;
					newProjectData.repositoryName = void 0;
					newProjectData.projectPathInsideRepository = void 0;
					newProjectData.vcsConfigurationPath = void 0;
					newProjectData.vcsProvider = void 0;
				}
			} else {
				const vcsData = await this.vcsManagementService.getProjectVCSData(projectPath);
				if (vcsData) {
					newProjectData.organizationName = vcsData.organizationName;
					newProjectData.repositoryName = vcsData.repositoryName;
					newProjectData.projectPathInsideRepository = vcsData.projectPathInsideRepository;
					newProjectData.vcsConfigurationPath = vcsData.vcsConfigurationPath;
					newProjectData.vcsProvider = vcsData.provider;
				} else {
					newProjectData.organizationName = void 0;
					newProjectData.repositoryName = void 0;
					newProjectData.projectPathInsideRepository = void 0;
					newProjectData.vcsConfigurationPath = void 0;
					newProjectData.vcsProvider = void 0;
				}
			}
			if (existingProjectData) {
				if (!newProjectData.cloudProjectId && existingProjectData.cloudProjectId) newProjectData.cloudProjectId = existingProjectData.cloudProjectId;
				if (!newProjectData.organizationId && existingProjectData.organizationId) newProjectData.organizationId = existingProjectData.organizationId;
				if (!newProjectData.genesisOrgId && existingProjectData.genesisOrgId) newProjectData.genesisOrgId = existingProjectData.genesisOrgId;
			}
			this.projects.set(projectPath, newProjectData);
			this.debouncedSaveAndBroadcast();
		} catch (error) {
			logger$28.warn("Failed to update project data on project file change", error);
		}
	}
	/**
	* Unified handler for all project file changes from the recursive root watcher.
	* Routes events to the appropriate handler based on relative path.
	*/
	async onProjectFileChange(projectPath, relativePath) {
		const normalized = relativePath.replaceAll("\\", "/");
		const handler = LocalProject.WATCHED_FILES[normalized];
		if (handler) {
			const fullPath = path.join(projectPath, relativePath);
			switch (handler) {
				case "settings":
					await this.onProjectSettingsChange(fullPath);
					break;
				case "renderPipeline":
					await this.onRenderPipelineFileChange(fullPath);
					break;
				case "buildTarget":
					await this.onBuildTargetFileChange(fullPath);
					break;
				default: break;
			}
			return;
		}
		if (LocalProject.VCS_EXACT_DIRS.has(normalized) || LocalProject.VCS_DIR_PREFIXES.some((p) => normalized.startsWith(p))) this.vcsManagementService.onVCSFileChanged(projectPath);
	}

	/**
	* Handles changes to the QualitySettings.asset and GraphicsSettings.asset files in ProjectSettings directory.

			}
		}
	}

	/**
	* Updates Unity Cloud project settings in a project's configuration files.
	*
	* This method modifies the ProjectSettings.asset file to set new Unity Cloud
	* project information including organization ID, cloud project ID, and project name.
	* It also updates the in-memory project data and saves changes to persistent storage.
	*
	* @param projectPath - The absolute path to the Unity project directory
	* @param unifiedProjectSettings - Object containing the new cloud project settings to apply
	* @throws {Error} Throws an error if the project data cannot be read or settings cannot be updated
	* @internal This method is used internally by cloud project linking operations
	*/
	async setUnifiedProjectSettings(projectPath, unifiedProjectSettings) {
		const projectData = this.projects.get(projectPath);
		if (!projectData) throw new Error("Could not read project data.");
		const projectSettingsFilePath = this.getProjectSettingsFilePath(projectPath, ProjectSettingsFileName);
		try {
			await updateCloudProjectData(projectSettingsFilePath, unifiedProjectSettings);
			projectData.cloudProjectId = unifiedProjectSettings.cloudProjectId;
			projectData.organizationId = unifiedProjectSettings.organizationId;
			projectData.genesisOrgId = unifiedProjectSettings.genesisOrgId;
			if (unifiedProjectSettings.projectName !== void 0) projectData.projectName = unifiedProjectSettings.projectName;
			this.projects.set(projectPath, projectData);
			await this.saveProjects();
		} catch (error) {
			logger$28.error("Failed to configure cloud project link.", error);
			throw new Error(`Failed to configure cloud project link. ${error}`);
		}
	}
	/**
	* Links a local Unity project to a Unity Cloud project.
	*
	* This method establishes a connection between a local Unity project and a Unity Cloud
	* project by updating the project's settings with cloud information. It validates the
	* cloud project ID format, updates the project configuration, sends analytics events,

	*
	* @param projectPath - The absolute path to the Unity project directory
	* @param organizationData - Object containing organization ID and genesis organization ID
	* @param cloudProjectId - The Unity Cloud project ID (must be a valid UUID)
	* @param projectName - The name to set for the project
	* @param existingCloudProjectFlag - Whether connecting to an existing cloud project (affects analytics)
	* @returns Promise that resolves when the linking operation is complete

	async linkCloudProject(projectPath, organizationData, cloudProjectId, projectName, existingCloudProjectFlag) {
		const { organizationId, genesisOrgId } = organizationData;
		const projectConnectedEventMessage = {
			organization_id: genesisOrgId,
			cloud_project_id: cloudProjectId,
			is_new_cloud_project: !existingCloudProjectFlag,
			is_new_local_project: false
		};
		try {
			if ((/* @__PURE__ */ new RegExp(/^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[1-5][0-9a-fA-F]{3}-[89abAB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}$/)).test(cloudProjectId) === false) throw new Error("Cloud project Id is not in valid uuid format");
			await this.setUnifiedProjectSettings(projectPath, {

				genesisOrgId
			});
			ProjectAnalyticsService.sendProjectConnectedEvent(projectConnectedEventMessage);

	/**
	* Disconnects a Unity project from Unity Cloud services.
	*
	* This method removes the Unity Cloud connection from a local project by clearing
	* cloud-related settings from the project configuration. It sends analytics events
	* about the disconnection and refreshes the project data to reflect the changes.
	*
	* @param project - The project object containing cloud connection information
	* @returns Promise that resolves when the unlinking operation is complete
	*/
	async unlinkCloudProject(project) {
		const { cloudProjectId, path: projectPath, genesisOrgId } = project;
		const projectDisconnectionEventMessage = {
			cloud_project_id: cloudProjectId,
			organization_id: genesisOrgId
		};
		try {
			await this.setUnifiedProjectSettings(projectPath, {
				organizationId: null,
				cloudProjectId: null,
				projectName: null,
				genesisOrgId: null
			});
			if (project.vcsProvider === SOURCE_CONTROL_PROVIDERS.UNITY_VERSION_CONTROL) {
				logger$28.info(`Disconnecting project from Unity Version Control: ${projectPath}`);
				await this.vcsManagementService.disconnectVersionControl(projectPath, this.updateProjectSettings.bind(this));
			}
			ProjectAnalyticsService.sendProjectDisconnectedEvent(projectDisconnectionEventMessage);
		} catch {
			ProjectAnalyticsService.sendProjectDisconnectedEvent(projectDisconnectionEventMessage);
		}
		await this.onProjectSettingsChange(this.getProjectSettingsFilePath(projectPath, ProjectSettingsFileName));
	}

	}
